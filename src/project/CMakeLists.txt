if(NOT TARGET nlohmann_json::nlohmann_json OR NOT TARGET Grapple::Http)
  message(FATAL_ERROR "The project CLI requires GRAPPLE_BEAM_WITH_JSON and GRAPPLE_BUILD_HTTP; disable GRAPPLE_BEAM_BUILD_APP for a library-only build")
endif()
include(templates.cmake)
set(embedded "#pragma once\n#include <string_view>\nnamespace grapple::project {\nstruct Template { const char *path; const char *text; };\ninline constexpr Template kTemplates[] = {\n")
foreach(path IN LISTS project_templates)
  set(source "${CMAKE_CURRENT_SOURCE_DIR}/templates/c/${path}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
  file(READ "${source}" content)
  string(APPEND embedded "{R\"GBFILE(${path})GBFILE\", R\"GBFILE(${content})GBFILE\"},\n")
endforeach()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/zlib.txt" zlib)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/MIT.txt" mit)
string(APPEND embedded "};\ninline constexpr const char *kZlibLicense = R\"GBFILE(${zlib})GBFILE\";\ninline constexpr const char *kMitLicense = R\"GBFILE(${mit})GBFILE\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/templates.hpp" "${embedded}")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
  OUTPUT_VARIABLE project_cli_commit OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git rev-parse --absolute-git-dir WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
  OUTPUT_VARIABLE project_git_dir OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(EXISTS "${project_git_dir}/HEAD")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${project_git_dir}/HEAD" "${project_git_dir}/index")
endif()
if(NOT project_cli_commit)
  set(project_cli_commit "v${PROJECT_VERSION}")
endif()
add_library(grapple_project STATIC cli.cpp create.cpp process.cpp subprocess.cpp package.cpp)
target_include_directories(grapple_project PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(grapple_project PRIVATE GRAPPLE_PROJECT_CLI_COMMIT="${project_cli_commit}")
target_link_libraries(grapple_project PRIVATE CLI11::CLI11 nlohmann_json::nlohmann_json Grapple::Http SDL3::SDL3)
grapple_set_project_warnings(grapple_project)
