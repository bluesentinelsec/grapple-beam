cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" GAME_VERSION LIMIT_COUNT 1)
project(@GAME_ID@ VERSION ${GAME_VERSION} LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
include(cmake/GrappleDependency.cmake)
include(cmake/Warnings.cmake)
enable_testing()
add_subdirectory(src)
add_subdirectory(tests)
include(cmake/Packaging.cmake)
if(EXISTS "${CMAKE_BINARY_DIR}/compile_commands.json")
  file(COPY_FILE "${CMAKE_BINARY_DIR}/compile_commands.json" "${CMAKE_SOURCE_DIR}/compile_commands.json" ONLY_IF_DIFFERENT)
endif()
