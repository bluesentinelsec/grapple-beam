add_executable(grapple_ttf_test
  ttf_test.cpp
  shaping_test.cpp
)

  PRIVATE
    Grapple::TTF
    GTest::gtest_main
)
# shaping_test.cpp probes the vendored HarfBuzz directly.

  ${CMAKE_SOURCE_DIR}/ttf/harfbuzz
)

  TTF_TEST_ASSETS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets"
)

gtest_discover_tests(grapple_ttf_test DISCOVERY_TIMEOUT 60)
